-- CUSTOM SERIALIZE (Vector3 → JSON friendly)
-- ═══════════════════════════════════════════════════════════════
local function serializePositions(positions)
    local out = {}
    if type(positions) ~= "table" then return out end
    for _, p in ipairs(positions) do
        if type(p) == "table" and p.pos then
            table.insert(out, {
                name = tostring(p.name or "Slot"),
                x = tonumber(p.pos.X) or 0,
                y = tonumber(p.pos.Y) or 0,
                z = tonumber(p.pos.Z) or 0,
            })
        end
    end
    return out
end

local function deserializePositions(data)
    local out = {}
    if type(data) ~= "table" then return out end
    for _, p in ipairs(data) do
        if type(p) == "table" and type(p.x) == "number" and type(p.y) == "number" and type(p.z) == "number" then
            table.insert(out, {
                name = tostring(p.name or "Slot"),
                pos = Vector3.new(p.x, p.y, p.z),
            })
        end
    end
    return out
end

-- CONFIG SAVE/LOAD
-- ═══════════════════════════════════════════════════════════════
local defaultConfig = {
    flySpeed = 60,
    guiX = 0.5,
    guiY = 0.5,
    savedPositions = {},
    slotCounter = 0,
}

local function saveConfig(data)
    if not writefile then return end
    local toSave = {}
    for k, v in pairs(data) do
        if k == "savedPositions" then
            toSave[k] = serializePositions(v)
        else
            toSave[k] = v
        end
    end
    local ok, encoded = pcall(function() return HttpService:JSONEncode(toSave) end)
    if ok and encoded then
        pcall(function() writefile(CONFIG.ConfigFile, encoded) end)
        print("[BSTS-ID] 💾 Config saved (" .. #(toSave.savedPositions or {}) .. " positions)")
    end
end

local function loadConfig()
    if not (isfile and readfile) then return defaultConfig end
    local ok, content = pcall(function()
        if isfile(CONFIG.ConfigFile) then return readfile(CONFIG.ConfigFile) end
    end)
    if ok and content and content ~= "" then
        local ok2, data = pcall(function() return HttpService:JSONDecode(content) end)
        if ok2 and type(data) == "table" then
            for k, v in pairs(defaultConfig) do
                if data[k] == nil then data[k] = v end
            end
            data.savedPositions = deserializePositions(data.savedPositions)
            print("[BSTS-ID] 📂 Config loaded (" .. #data.savedPositions .. " positions)")
            return data
        end
    end
    return defaultConfig
end

local userConfig = loadConfig()

-- GAME CHECK
-- ═══════════════════════════════════════════════════════════════
local function isBSTSGame()
    for _, id in ipairs(CONFIG.AllowedPlaceIds) do
        if game.PlaceId == id then return true end
    end
    local nameLower = string.lower(game.Name)
    for _, kw in ipairs(CONFIG.AllowedNameKeywords) do
        if string.find(nameLower, kw, 1, true) then return true end
    end
    return false
end

    local function findNearestCar()
        local char = LocalPlayer.Character
        if not char then return nil end
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not hrp then return nil end
        local nearestSeat, shortestDist = nil, math.huge
        for _, obj in pairs(workspace:GetDescendants()) do
            if obj:IsA("VehicleSeat") then
                local dist = (obj.Position - hrp.Position).Magnitude
                if dist < shortestDist then
                    shortestDist = dist
                    nearestSeat = obj
                end
            end
        end
        if not nearestSeat then return nil end
        local vModel = nearestSeat:FindFirstAncestorOfClass("Model") or nearestSeat.Parent
        local root = vModel
        local cur = vModel
        while cur and cur.Parent and cur.Parent ~= workspace do
            cur = cur.Parent
            if cur:IsA("Model") then root = cur end
        end
        return root
    end

    local function teleportCarToPos(pos)
        if not pos or not targetVehicleRoot then return end
        local primaryPart = targetVehicleRoot.PrimaryPart or targetVehicleRoot:FindFirstChildWhichIsA("BasePart")
        if not primaryPart then
            for _, d in pairs(targetVehicleRoot:GetDescendants()) do
                if d:IsA("BasePart") then
                    primaryPart = d
                    break
                end
            end
        end
        if not primaryPart then return end

        local allParts, relCF = {}, {}
        for _, p in pairs(targetVehicleRoot:GetDescendants()) do
            if p:IsA("BasePart") then
                table.insert(allParts, p)
            end
        end
        for _, p in pairs(allParts) do
            relCF[p] = primaryPart.CFrame:ToObjectSpace(p.CFrame)
        end
        for _, p in pairs(allParts) do
            p.Anchored = true
        end

        primaryPart.CFrame = CFrame.new(pos + Vector3.new(0, 3, 0))
        for p, rel in pairs(relCF) do
            p.CFrame = primaryPart.CFrame:ToWorldSpace(rel)
        end

        task.wait(0.1)
        for _, p in pairs(allParts) do
            p.Anchored = false
            pcall(function()
                if p:CanSetNetworkOwnership() then
                    p:SetNetworkOwner(LocalPlayer)
                end
            end)
        end
        for _, d in pairs(targetVehicleRoot:GetDescendants()) do
            if d:IsA("LocalScript") and not d.Enabled then
                d.Enabled = true
            end
        end
    end

    -- ═══ REFRESH LIST (FIXED pakai typeof) ═══
    refreshPositionList = function()
        -- Bersihin entry corrupt (pakai typeof!)
        for i = #savedPositions, 1, -1 do
            local pd = savedPositions[i]
            if type(pd) ~= "table" or typeof(pd.pos) ~= "Vector3" or type(pd.name) ~= "string" then
                table.remove(savedPositions, i)
                markSaveDirty()
                print("[BSTS-ID] 🗑 Hapus entry corrupt index " .. i)
            end
        end

        for _, c in pairs(PositionsList:GetChildren()) do
            if c:IsA("Frame") then
                c:Destroy()
            end
        end

        PosTitle.Text = "💾  POSISI TERSIMPAN (" .. #savedPositions .. ")"

        if #savedPositions == 0 then
            EmptyLabel.Visible = true
        else
            EmptyLabel.Visible = false
            for i, pd in ipairs(savedPositions) do
                local ok, err = pcall(buildSlotRow, pd, i)
                if not ok then
                    warn("[BSTS-ID] buildSlotRow error: " .. tostring(err))
                end
            end
        end

        -- Set CanvasSize manual
        local contentHeight = 0
        if #savedPositions == 0 then
            contentHeight = 72
        else
            contentHeight = #savedPositions * (ROW_HEIGHT + ROW_PADDING) + 12
        end
        PositionsList.CanvasSize = UDim2.new(0, 0, 0, contentHeight)

        print("[BSTS-ID] 📋 Refresh list: " .. #savedPositions .. " positions, canvas: " .. contentHeight)
    end

    -- ═══ OPEN/CLOSE ANIMATION (SIZE + FADE) ═══
    local isAnimating = false

    local function openGUI()
        if MainFrame.Visible then return end
        if isAnimating then return end
        isAnimating = true

        MainFrame.Visible = true
        MainFrame.Size = UDim2.new(0, 0, 0, 0)
        MainFrame.BackgroundTransparency = 1
        MainStroke.Transparency = 1

        pcall(function()
            local tw = tween(MainFrame, {Size = UDim2.new(0, guiW, 0, guiH)}, 0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
            tw:Play()
            tween(MainFrame, {BackgroundTransparency = 0}, 0.3):Play()
            tween(MainStroke, {Transparency = 0.4}, 0.3):Play()
            tween(ToggleBtn, {Rotation = 90}, 0.35):Play()
        end)

        task.spawn(function()
            task.wait(0.8)
            if MainFrame.Visible and MainFrame.Size ~= UDim2.new(0, guiW, 0, guiH) then
                MainFrame.Size = UDim2.new(0, guiW, 0, guiH)
                MainFrame.BackgroundTransparency = 0
                MainStroke.Transparency = 0.4
            end
        end)

        task.wait(0.4)
        isAnimating = false
    end

    local function closeGUI()
        if not MainFrame.Visible then return end
        if isAnimating then return end
        isAnimating = true

        pcall(function()
            local tw = tween(MainFrame, {Size = UDim2.new(0, 0, 0, 0)}, 0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
            tw:Play()
            tween(MainFrame, {BackgroundTransparency = 1}, 0.2):Play()
            tween(MainStroke, {Transparency = 1}, 0.2):Play()
            tween(ToggleBtn, {Rotation = 0}, 0.3):Play()
        end)

        task.wait(0.3)
        MainFrame.Visible = false
        MainFrame.Size = UDim2.new(0, guiW, 0, guiH)
        MainFrame.BackgroundTransparency = 0
        MainStroke.Transparency = 0.4
        isAnimating = false
    end

    local function toggleGUI()
        if MainFrame.Visible then
            closeGUI()
        else
            openGUI()
        end
    end

    bind(CloseBtn.MouseButton1Click, closeGUI)

        else

            FlyBtn.BackgroundColor3 = Color3.fromRGB(30, 30, 40)
            StatusLabel.Text = "Status: Fly nonaktif"
        end
    end)

    bind(FlySpeedBox.FocusLost, function()
        local v = tonumber(FlySpeedBox.Text)
        if v and v > 0 and v <= 500 then
            flySpeed = v
            markSaveDirty()
            StatusLabel.Text = "Status: Fly speed → " .. v
        else
            FlySpeedBox.Text = tostring(flySpeed)
            StatusLabel.Text = "⚠️ Fly speed harus 1-500"
        end
    end)

    bind(UserInputService.InputBegan, function(input, gp)
        if gp then return end
        if input.KeyCode == Enum.KeyCode.T then
            if not targetVehicleRoot then
                targetVehicleRoot = findNearestCar()
            end
            if #savedPositions > 0 then
                teleportCarToPos(savedPositions[#savedPositions].pos)
            end
        elseif input.KeyCode == Enum.KeyCode.F then
            toggleFly()
            if flyEnabled then
                FlyLbl.Text = "✈️  Fly: ON"
                FlyBtn.BackgroundColor3 = COLOR.Green1
            else
                FlyLbl.Text = "✈️  Fly: OFF"
                FlyBtn.BackgroundColor3 = Color3.fromRGB(30, 30, 40)
            end
        elseif input.KeyCode == Enum.KeyCode.RightControl then
            toggleGUI()
        end
    end)

    print("[BSTS-ID] ✅ Car Tool siap!")
end
