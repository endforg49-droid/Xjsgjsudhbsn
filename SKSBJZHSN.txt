local function createAnimatedBorder(parent, cornerRadius, thickness)
    thickness = thickness or 3
    local offset = thickness
    local border = Instance.new("Frame")
    border.Name = "AnimBorder"
    border.Size = UDim2.new(1, offset * 2, 1, offset * 2)
    border.Position = UDim2.new(0, -offset, 0, -offset)
    border.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    border.BorderSizePixel = 0
    border.ZIndex = (parent.ZIndex or 0) - 1
    border.Parent = parent
    local bc = Instance.new("UICorner")
    bc.CornerRadius = UDim.new(0, (cornerRadius or 18) + offset)
    bc.Parent = border
    local grad = Instance.new("UIGradient")
    grad.Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0.00, Color3.fromRGB(255, 255, 255)),
        ColorSequenceKeypoint.new(0.15, Color3.fromRGB(80, 80, 80)),
        ColorSequenceKeypoint.new(0.30, Color3.fromRGB(0, 0, 0)),

-- ═══ SERIALIZE ═══
local function serializePositions(positions)
    local out = {}
    if type(positions) ~= "table" then return out end
    for _, p in ipairs(positions) do
        if type(p) == "table" and p.pos then
            table.insert(out, {name = tostring(p.name or "Slot"), x = tonumber(p.pos.X) or 0, y = tonumber(p.pos.Y) or 0, z = tonumber(p.pos.Z) or 0})
        end
    end
    return out
end
local function deserializePositions(data)
    local out = {}
    if type(data) ~= "table" then return out end
    for _, p in ipairs(data) do
        if type(p) == "table" and type(p.x) == "number" and type(p.y) == "number" and type(p.z) == "number" then
            table.insert(out, {name = tostring(p.name or "Slot"), pos = Vector3.new(p.x, p.y, p.z)})
        end
    end
    return out
end

local defaultConfig = {

}

local function saveConfig(data)
    if not writefile then return end
    local toSave = {}
    for k, v in pairs(data) do
        if k == "savedPositions" then toSave[k] = serializePositions(v)
        else toSave[k] = v end
    end
    local ok, encoded = pcall(function() return HttpService:JSONEncode(toSave) end)
    if ok and encoded then pcall(function() writefile(CONFIG.ConfigFile, encoded) end) end
end

local function loadConfig()
    if not (isfile and readfile) then return defaultConfig end
    local ok, content = pcall(function()
        if isfile(CONFIG.ConfigFile) then return readfile(CONFIG.ConfigFile) end
    end)
    if ok and content and content ~= "" then
        local ok2, data = pcall(function() return HttpService:JSONDecode(content) end)
        if ok2 and type(data) == "table" then
            for k, v in pairs(defaultConfig) do
                if data[k] == nil then data[k] = v end
            end
            data.savedPositions = deserializePositions(data.savedPositions)
            return data
        end
    end
    return defaultConfig
end

    local function findNearestCar()
        local char = LocalPlayer.Character
        if not char then return nil end
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not hrp then return nil end
        local nearestSeat, shortestDist = nil, math.huge
        for _, obj in pairs(workspace:GetDescendants()) do
            if obj:IsA("VehicleSeat") then
                local dist = (obj.Position - hrp.Position).Magnitude
                if dist < shortestDist then shortestDist = dist; nearestSeat = obj end
            end
        end
        if not nearestSeat then return nil end
        local vModel = nearestSeat:FindFirstAncestorOfClass("Model") or nearestSeat.Parent
        local root = vModel
        local cur = vModel
        while cur and cur.Parent and cur.Parent ~= workspace do
            cur = cur.Parent
            if cur:IsA("Model") then root = cur end
        end
        return root
    end

    local function collectAllNames(instance, maxDepth)
        local names = {}
        maxDepth = maxDepth or 15
        local cur = instance
        local depth = 0
        while cur and cur ~= game and depth < maxDepth do
            depth = depth + 1
            pcall(function()
                if cur.Name and cur.Name ~= "" then
                    table.insert(names, string.lower(cur.Name))
                end
            end)
            pcall(function()
                if cur.DisplayName and cur.DisplayName ~= "" then
                    table.insert(names, string.lower(cur.DisplayName))
                end
            end)
            pcall(function()
                local attrs = cur:GetAttributes()
                for attrName, attrVal in pairs(attrs) do
                    if type(attrVal) == "string" then
                        table.insert(names, string.lower(attrVal))
                    end
                    table.insert(names, string.lower(attrName))
                end
            end)
            cur = cur.Parent
        end
        return names
    end

    local function getDrivenVehicle()

        if not seat then return nil, {} end
        local names = collectAllNames(seat, 15)
        local topModel = seat
        local cur = seat
        while cur and cur.Parent and cur.Parent ~= workspace do
            cur = cur.Parent
            if cur:IsA("Model") then topModel = cur end
        end
        return topModel, names
    end

    local function isDrivingFarmTruck()
        local vehicle, names = getDrivenVehicle()
        if #names == 0 then
            return false, nil, "Gak duduk di mobil"
        end
        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        local seat = hum and hum.SeatPart
        if seat then
            table.insert(names, string.lower(seat.Name or ""))
        end
        local kwLower = string.lower(CONFIG.TruckKeyword)
        local kwParts = {}
        for part in string.gmatch(kwLower, "%S+") do
            table.insert(kwParts, part)
        end
        for _, name in ipairs(names) do
            if string.find(name, kwLower, 1, true) then
                return true, vehicle, "Match full: " .. name
            end
        end
        for _, name in ipairs(names) do
            local allMatch = true
            for _, part in ipairs(kwParts) do
                if not string.find(name, part, 1, true) then
                    allMatch = false
                    break
                end
            end
            if allMatch and #kwParts > 0 then
                return true, vehicle, "Match part: " .. name
            end
        end
        return false, vehicle, "Gak match keyword"
    end

    local function teleportCarToPos(pos)
        if not pos then return end
        local vehicle = targetVehicleRoot or getDrivenVehicle()
        if not vehicle then return end
        local primaryPart = vehicle.PrimaryPart or vehicle:FindFirstChildWhichIsA("BasePart")
        if not primaryPart then
            for _, d in pairs(vehicle:GetDescendants()) do
                if d:IsA("BasePart") then primaryPart = d break end
            end
        end
        if not primaryPart then return end
        local allParts, relCF = {}, {}
        for _, p in pairs(vehicle:GetDescendants()) do
            if p:IsA("BasePart") then table.insert(allParts, p) end
        end
        for _, p in pairs(allParts) do relCF[p] = primaryPart.CFrame:ToObjectSpace(p.CFrame) end
        for _, p in pairs(allParts) do p.Anchored = true end
        primaryPart.CFrame = CFrame.new(pos + Vector3.new(0, 3, 0))
        for p, rel in pairs(relCF) do p.CFrame = primaryPart.CFrame:ToWorldSpace(rel) end
        task.wait(0.1)
        for _, p in pairs(allParts) do
            p.Anchored = false
            pcall(function()
                if p:CanSetNetworkOwnership() then p:SetNetworkOwner(LocalPlayer) end
            end)
        end
    end

        keyword = string.lower(keyword)

        local found = {}
        for _, root in ipairs(roots) do

                        end
                    end
                end
            end
        end
        return found
    end
